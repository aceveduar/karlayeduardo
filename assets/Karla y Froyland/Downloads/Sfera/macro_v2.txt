Sub Paso1_PrepararDatos()
    Dim ws As Worksheet
    Set ws = ActiveSheet
    Dim LastRow As Long, LastCol As Long
    Dim c As Long, r As Long
    Dim keepCols As Variant, headerVal As String, isKept As Boolean
    
    ' Acelerar la macro y evitar alertas visuales
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' 1. QUITAR FILTROS DE FORMA SEGURA (Por si el archivo ya traía)
    On Error Resume Next
    ws.AutoFilterMode = False
    If ws.FilterMode = True Then ws.ShowAllData
    Dim lst As ListObject
    For Each lst In ws.ListObjects
        lst.AutoFilter.ShowAllData
    Next lst
    On Error GoTo 0
    
    ' 2. ENCONTRAR Y ELIMINAR LA FILA DE TOTALES ORIGINAL
    Dim totalCell As Range
    Set totalCell = ws.UsedRange.find(What:="total", LookIn:=xlValues, LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False)
    If Not totalCell Is Nothing Then ws.Rows(totalCell.Row).Delete
    
    LastRow = ws.cells(ws.Rows.Count, "A").End(xlUp).Row
    LastCol = ws.cells(1, ws.Columns.Count).End(xlToLeft).Column
    
    ' 3. DEJAR ÚNICAMENTE LAS COLUMNAS SOLICITADAS
    keepCols = Array("Fecha", "Nombre de la campaña", "Categoría de la campaña/acción", _
                     "Código de operación", "Asunto", "Destinatarios", "Exitosos", _
                     "% Exitosos", "Hard bounces", "% Hard bounces", "Aperturas", _
                     "% Aperturas", "Abridores únicos", "% Abridores únicos", "Clics", _
                     "% Clics", "Clickers únicos", "% Clickers únicos", "% Click-To-Open Rate", _
                     "% Tasa de clics por aperturas", "Bajas", "% Bajas", "Quejas por spam", _
                     "% Quejas por spam")
                     
    For c = LastCol To 1 Step -1
        headerVal = Trim(ws.cells(1, c).value)
        isKept = False
        Dim i As Integer
        For i = LBound(keepCols) To UBound(keepCols)
            If LCase(headerVal) = LCase(keepCols(i)) Then
                isKept = True
                Exit For
            End If
        Next i
        If Not isKept Then ws.Columns(c).Delete
    Next c
    
    LastCol = ws.cells(1, ws.Columns.Count).End(xlToLeft).Column
    
    ' 4. MAPEAR COLUMNAS
    Dim dictCols As Object
    Set dictCols = CreateObject("Scripting.Dictionary")
    For c = 1 To LastCol
        dictCols.Add LCase(Trim(ws.cells(1, c).value)), c
    Next c
    
    ' --- INSERTAR COLUMNA "MES Y AÑO" DE FORMA SEGURA ---
    If dictCols.exists("fecha") Then
        ws.Columns(dictCols("fecha") + 1).Insert Shift:=xlToRight
        ws.cells(1, dictCols("fecha") + 1).value = "Mes y Año"
        LastCol = LastCol + 1
        ' Re-mapear columnas por el desplazamiento
        Set dictCols = CreateObject("Scripting.Dictionary")
        For c = 1 To LastCol
            dictCols.Add LCase(Trim(ws.cells(1, c).value)), c
        Next c
    End If
    
    ' --- LIMPIEZA DE FECHAS Y ORDENAMIENTO ---
    If dictCols.exists("fecha") Then
        For r = 2 To LastRow
            Dim strDate As String
            strDate = Trim(CStr(ws.cells(r, dictCols("fecha")).value))
            
            ' Quitamos la hora sin empujar ni borrar otras columnas
            If InStr(strDate, " ") > 0 Then
                strDate = Split(strDate, " ")(0)
                ws.cells(r, dictCols("fecha")).value = strDate
            End If
            
            ' Llenamos el Mes y Año
            If dictCols.exists("mes y año") And IsDate(strDate) Then
                ws.cells(r, dictCols("mes y año")).value = UCase(Left(Format(CDate(strDate), "mmmm yyyy"), 1)) & Mid(Format(CDate(strDate), "mmmm yyyy"), 2)
            End If
        Next r
        
        ' Ordenamos cronológicamente
        ws.Sort.SortFields.Clear
        ws.Sort.SortFields.Add key:=ws.cells(2, dictCols("fecha")), SortOn:=xlSortOnValues, Order:=xlDescending, DataOption:=xlSortNormal
        With ws.Sort
            .SetRange ws.Range(ws.cells(1, 1), ws.cells(LastRow, LastCol))
            .header = xlYes
            .Apply
        End With
    End If
    
    ' 5. APLICAR FÓRMULAS DE PORCENTAJES
    For r = 2 To LastRow
        If dictCols.exists("% exitosos") Then ws.cells(r, dictCols("% exitosos")).Formula = "=IFERROR(" & ws.cells(r, dictCols("exitosos")).Address(False, False) & "/" & ws.cells(r, dictCols("destinatarios")).Address(False, False) & ", 0)"
        If dictCols.exists("% hard bounces") Then ws.cells(r, dictCols("% hard bounces")).Formula = "=IFERROR(" & ws.cells(r, dictCols("hard bounces")).Address(False, False) & "/" & ws.cells(r, dictCols("destinatarios")).Address(False, False) & ", 0)"
        If dictCols.exists("% aperturas") Then ws.cells(r, dictCols("% aperturas")).Formula = "=IFERROR(" & ws.cells(r, dictCols("aperturas")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        If dictCols.exists("% abridores únicos") Then ws.cells(r, dictCols("% abridores únicos")).Formula = "=IFERROR(" & ws.cells(r, dictCols("abridores únicos")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        If dictCols.exists("% clics") Then ws.cells(r, dictCols("% clics")).Formula = "=IFERROR(" & ws.cells(r, dictCols("clics")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        If dictCols.exists("% clickers únicos") Then ws.cells(r, dictCols("% clickers únicos")).Formula = "=IFERROR(" & ws.cells(r, dictCols("clickers únicos")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        If dictCols.exists("% click-to-open rate") Then ws.cells(r, dictCols("% click-to-open rate")).Formula = "=IFERROR(" & ws.cells(r, dictCols("clickers únicos")).Address(False, False) & "/" & ws.cells(r, dictCols("abridores únicos")).Address(False, False) & ", 0)"
        If dictCols.exists("% tasa de clics por aperturas") Then ws.cells(r, dictCols("% tasa de clics por aperturas")).Formula = "=IFERROR(" & ws.cells(r, dictCols("clics")).Address(False, False) & "/" & ws.cells(r, dictCols("aperturas")).Address(False, False) & ", 0)"
        If dictCols.exists("% bajas") Then ws.cells(r, dictCols("% bajas")).Formula = "=IFERROR(" & ws.cells(r, dictCols("bajas")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        If dictCols.exists("% quejas por spam") Then ws.cells(r, dictCols("% quejas por spam")).Formula = "=IFERROR(" & ws.cells(r, dictCols("quejas por spam")).Address(False, False) & "/" & ws.cells(r, dictCols("exitosos")).Address(False, False) & ", 0)"
        
        Dim colName As Variant
        For Each colName In Array("% exitosos", "% hard bounces", "% aperturas", "% abridores únicos", "% clics", "% clickers únicos", "% click-to-open rate", "% tasa de clics por aperturas", "% bajas", "% quejas por spam")
            If dictCols.exists(colName) Then ws.cells(r, dictCols(colName)).NumberFormat = "0.00%"
        Next colName
    Next r
    
    ' 6. SACAR NUEVOS TOTALES DINÁMICOS
    Dim TotalRow As Long
    TotalRow = LastRow + 2
    ws.cells(TotalRow, 1).value = "TOTALES"
    
    Dim colsToSum As Variant
    colsToSum = Array("destinatarios", "exitosos", "hard bounces", "aperturas", "abridores únicos", "clics", "clickers únicos", "bajas", "quejas por spam")
    For Each colName In colsToSum
        If dictCols.exists(colName) Then
            ws.cells(TotalRow, dictCols(colName)).Formula = "=SUBTOTAL(109," & ws.cells(2, dictCols(colName)).Address(False, False) & ":" & ws.cells(LastRow, dictCols(colName)).Address(False, False) & ")"
        End If
    Next colName
    
    Dim colsToCalc As Variant
    colsToCalc = Array("% exitosos", "% hard bounces", "% aperturas", "% abridores únicos", "% clics", "% clickers únicos", "% click-to-open rate", "% tasa de clics por aperturas", "% bajas", "% quejas por spam")
    For Each colName In colsToCalc
        If dictCols.exists(colName) Then
            c = dictCols(colName)
            Dim numCol As Long, denCol As Long
            Select Case colName
                Case "% exitosos", "% hard bounces": denCol = dictCols("destinatarios"): If colName = "% exitosos" Then numCol = dictCols("exitosos") Else numCol = dictCols("hard bounces")
                Case "% aperturas", "% abridores únicos", "% clics", "% clickers únicos", "% bajas", "% quejas por spam": denCol = dictCols("exitosos"): If colName = "% aperturas" Then numCol = dictCols("aperturas") Else If colName = "% abridores únicos" Then numCol = dictCols("abridores únicos") Else If colName = "% clics" Then numCol = dictCols("clics") Else If colName = "% clickers únicos" Then numCol = dictCols("clickers únicos") Else If colName = "% bajas" Then numCol = dictCols("bajas") Else numCol = dictCols("quejas por spam")
                Case "% click-to-open rate": numCol = dictCols("clickers únicos"): denCol = dictCols("abridores únicos")
                Case "% tasa de clics por aperturas": numCol = dictCols("clics"): denCol = dictCols("aperturas")
            End Select
            ws.cells(TotalRow, c).Formula = "=IFERROR(" & ws.cells(TotalRow, numCol).Address(False, False) & "/" & ws.cells(TotalRow, denCol).Address(False, False) & ", 0)"
            ws.cells(TotalRow, c).NumberFormat = "0.00%"
        End If
    Next colName
    
    ' SEPARADOR DE MILES
    For Each colName In colsToSum
        If dictCols.exists(colName) Then ws.Range(ws.cells(2, dictCols(colName)), ws.cells(TotalRow, dictCols(colName))).NumberFormat = "#,##0"
    Next colName

    ' FORMATO CONDICIONAL
    Dim fc As FormatCondition, cs As ColorScale, rngFC As Range
    For Each colName In Array("% hard bounces", "% bajas", "% quejas por spam")
        If dictCols.exists(colName) Then
            Set rngFC = ws.Range(ws.cells(2, dictCols(colName)), ws.cells(LastRow, dictCols(colName)))
            rngFC.FormatConditions.Delete
            Set fc = rngFC.FormatConditions.Add(Type:=xlCellValue, Operator:=xlGreater, Formula1:="=0.01")
            fc.Interior.Color = RGB(255, 199, 206): fc.Font.Color = RGB(156, 0, 6)
        End If
    Next colName
    For Each colName In Array("% aperturas", "% tasa de clics por aperturas", "% click-to-open rate")
        If dictCols.exists(colName) Then
            Set rngFC = ws.Range(ws.cells(2, dictCols(colName)), ws.cells(LastRow, dictCols(colName)))
            rngFC.FormatConditions.Delete
            Set cs = rngFC.FormatConditions.AddColorScale(ColorScaleType:=2)
            cs.ColorScaleCriteria(1).Type = xlConditionValueLowestValue: cs.ColorScaleCriteria(1).FormatColor.Color = RGB(255, 255, 255)
            cs.ColorScaleCriteria(2).Type = xlConditionValueHighestValue: cs.ColorScaleCriteria(2).FormatColor.Color = RGB(198, 239, 206)
        End If
    Next colName

    ' 7. FORMATO VISUAL
    ws.cells.EntireColumn.AutoFit
    For c = 1 To LastCol
        If ws.Columns(c).ColumnWidth > 25 Then ws.Columns(c).ColumnWidth = 25
    Next c
    ws.Rows("2:" & LastRow).WrapText = False
    
    With ws.Range(ws.cells(1, 1), ws.cells(1, LastCol))
        .WrapText = True: .HorizontalAlignment = xlCenter: .VerticalAlignment = xlCenter
        .Font.Bold = True: .Interior.Color = RGB(220, 230, 241)
    End With
    ws.Rows(1).AutoFit
    
    With ws.Range(ws.cells(TotalRow, 1), ws.cells(TotalRow, LastCol))
        .Font.Bold = True: .Interior.Color = RGB(255, 242, 204)
        .Borders(xlEdgeTop).LineStyle = xlContinuous: .Borders(xlEdgeBottom).LineStyle = xlDouble: .Borders(xlEdgeBottom).Weight = xlThick
    End With
    
    ws.Activate
    ws.Rows(2).Select
    ActiveWindow.FreezePanes = True
    ws.cells(1, 1).Select
    
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    
    MsgBox "¡Datos preparados y limpios!" & vbCrLf & vbCrLf & "Puedes aplicar tus propios filtros a las columnas desde el menú 'Datos > Filtro' cuando lo necesites." & vbCrLf & "Cuando termines de seleccionar tus datos, ejecuta 'Paso2_GenerarDashboard'.", vbInformation, "Paso 1 Completado"
End Sub

Sub Paso2_GenerarDashboard()
    Dim wsData As Worksheet, wsResumen As Worksheet, wsTemp As Worksheet
    Set wsData = ActiveSheet
    
    ' Validar que estamos en la hoja de datos
    Dim cDest As Range
    Set cDest = wsData.Rows(1).find(What:="Destinatarios", LookAt:=xlWhole, MatchCase:=False)
    If cDest Is Nothing Then
        MsgBox "Por favor, colócate en tu hoja de datos filtrada para generar el Dashboard.", vbExclamation, "Error"
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    On Error Resume Next
    Sheets("Dashboard Ejecutivo").Delete
    On Error GoTo 0
    
    ' Crear hoja temporal con lo visible
    Set wsTemp = Sheets.Add
    wsData.UsedRange.SpecialCells(xlCellTypeVisible).Copy wsTemp.Range("A1")
    
    Dim LastRow As Long, LastCol As Long
    LastRow = wsTemp.cells(wsTemp.Rows.Count, "A").End(xlUp).Row
    LastCol = wsTemp.cells(1, wsTemp.Columns.Count).End(xlToLeft).Column
    
    ' Borrar fila de totales temporal
    Dim r As Long
    For r = LastRow To 2 Step -1
        If UCase(wsTemp.cells(r, 1).value) = "TOTALES" Then
            wsTemp.Rows(r).Delete
            LastRow = LastRow - 1
            Exit For
        End If
    Next r
    
    If LastRow < 2 Then
        Application.DisplayAlerts = False
        wsTemp.Delete
        Application.DisplayAlerts = True
        MsgBox "No hay datos visibles para generar el reporte.", vbExclamation, "Sin datos"
        Exit Sub
    End If
    
    Dim dict As Object: Set dict = CreateObject("Scripting.Dictionary")
    Dim c As Long
    For c = 1 To LastCol: dict.Add LCase(Trim(wsTemp.cells(1, c).value)), c: Next c
    
    Set wsResumen = Sheets.Add(before:=Sheets(1))
    wsResumen.Name = "Dashboard Ejecutivo"
    
    ' =======================================================
    ' 1. CONFIGURACIÓN VISUAL BASE (ESTILO WEB)
    ' =======================================================
    ActiveWindow.DisplayGridlines = False
    wsResumen.cells.Font.Name = "Segoe UI"
    wsResumen.cells.Interior.Color = RGB(241, 245, 249) ' Fondo Gris Web (Slate-50)
    
    wsResumen.cells(2, 2).value = "?? DASHBOARD EJECUTIVO DE CAMPAÑAS"
    wsResumen.cells(2, 2).Font.Size = 22
    wsResumen.cells(2, 2).Font.Bold = True
    wsResumen.cells(2, 2).Font.Color = RGB(30, 41, 59) ' Slate-800
    
    ' =======================================================
    ' 2. SCORECARDS HORIZONTALES (SALUD GENERAL)
    ' =======================================================
    Dim totDest As Double, totExi As Double, totApe As Double, totClic As Double, totBajas As Double
    If dict.exists("destinatarios") Then totDest = Application.WorksheetFunction.Sum(wsTemp.Columns(dict("destinatarios")))
    If dict.exists("exitosos") Then totExi = Application.WorksheetFunction.Sum(wsTemp.Columns(dict("exitosos")))
    If dict.exists("aperturas") Then totApe = Application.WorksheetFunction.Sum(wsTemp.Columns(dict("aperturas")))
    If dict.exists("clics") Then totClic = Application.WorksheetFunction.Sum(wsTemp.Columns(dict("clics")))
    If dict.exists("bajas") Then totBajas = Application.WorksheetFunction.Sum(wsTemp.Columns(dict("bajas")))
    
    ' Formato de la tarjeta base
    Dim rngScore As Range
    Set rngScore = wsResumen.Range("B4:G5")
    rngScore.Interior.Color = RGB(255, 255, 255)
    rngScore.BorderAround xlContinuous, xlThin, , RGB(203, 213, 225)
    With wsResumen.Range("B4:G4").Borders(xlEdgeTop)
        .LineStyle = xlContinuous: .Weight = xlThick: .Color = RGB(37, 99, 235) ' Azul Brillante
    End With
    
    ' Textos
    wsResumen.cells(4, 2).value = "ENVÍOS EXITOSOS"
    wsResumen.cells(4, 3).value = "APERTURA PROMEDIO"
    wsResumen.cells(4, 5).value = "CTR (CLICS/ENVÍOS)"
    wsResumen.cells(4, 6).value = "CTOR (INTERACCIÓN)"
    wsResumen.cells(4, 7).value = "TASA DE BAJAS"
    
    ' Valores
    wsResumen.cells(5, 2).value = totExi: wsResumen.cells(5, 2).NumberFormat = "#,##0"
    If totExi > 0 Then wsResumen.cells(5, 3).value = totApe / totExi Else wsResumen.cells(5, 3).value = 0
    If totExi > 0 Then wsResumen.cells(5, 5).value = totClic / totExi Else wsResumen.cells(5, 5).value = 0
    If totApe > 0 Then wsResumen.cells(5, 6).value = totClic / totApe Else wsResumen.cells(5, 6).value = 0
    If totExi > 0 Then wsResumen.cells(5, 7).value = totBajas / totExi Else wsResumen.cells(5, 7).value = 0
    wsResumen.Range("C5, E5:G5").NumberFormat = "0.00%"
    
    ' Alineación
    With wsResumen.Range("B4:G4")
        .Font.Size = 9: .Font.Bold = True: .Font.Color = RGB(100, 116, 139)
        .HorizontalAlignment = xlCenter: .VerticalAlignment = xlCenter
    End With
    With wsResumen.Range("B5:G5")
        .Font.Size = 16: .Font.Bold = True: .Font.Color = RGB(15, 23, 42)
        .HorizontalAlignment = xlCenter: .VerticalAlignment = xlCenter
    End With
    
    ' Altura de filas
    wsResumen.Rows(4).RowHeight = 22
    wsResumen.Rows(5).RowHeight = 35
    
    ' Punteros de filas dinámicos para columnas Izquierda y Derecha
    Dim rowL As Long, rowR As Long
    rowL = 7
    rowR = 7
    
    ' =======================================================
    ' 3. TARJETA IZQUIERDA 1: CATEGORÍAS
    ' =======================================================
    If dict.exists("categoría de la campaña/acción") Then
        Dim dictCatExi As Object: Set dictCatExi = CreateObject("Scripting.Dictionary")
        Dim dictCatApe As Object: Set dictCatApe = CreateObject("Scripting.Dictionary")
        
        For r = 2 To LastRow
            Dim cat As String: cat = wsTemp.cells(r, dict("categoría de la campaña/acción")).value
            If cat = "" Then cat = "Sin Categoría"
            dictCatExi(cat) = dictCatExi(cat) + wsTemp.cells(r, dict("exitosos")).value
            dictCatApe(cat) = dictCatApe(cat) + wsTemp.cells(r, dict("aperturas")).value
        Next r
        
        wsResumen.cells(rowL, 2).value = "??? RENDIMIENTO POR CATEGORÍA"
        wsResumen.cells(rowL, 3).value = "Apertura %"
        
        Dim catKey As Variant, startCat As Long
        startCat = rowL + 1
        For Each catKey In dictCatExi.keys
            wsResumen.cells(startCat, 2).value = catKey
            If dictCatExi(catKey) > 0 Then wsResumen.cells(startCat, 3).value = dictCatApe(catKey) / dictCatExi(catKey) Else wsResumen.cells(startCat, 3).value = 0
            startCat = startCat + 1
        Next catKey
        
        ' Formato de Tarjeta
        Dim rngCat As Range: Set rngCat = wsResumen.Range(wsResumen.cells(rowL, 2), wsResumen.cells(startCat - 1, 3))
        rngCat.Interior.Color = RGB(255, 255, 255)
        rngCat.BorderAround xlContinuous, xlThin, , RGB(203, 213, 225)
        
        With wsResumen.Range(wsResumen.cells(rowL, 2), wsResumen.cells(rowL, 3))
            .Interior.Color = RGB(30, 41, 59) ' Dark Slate
            .Font.Color = RGB(255, 255, 255): .Font.Bold = True: .Font.Size = 11
            .VerticalAlignment = xlCenter
        End With
        
        ' Ordenar y cebrado
        wsResumen.Sort.SortFields.Clear
        wsResumen.Sort.SortFields.Add key:=wsResumen.Range(wsResumen.cells(rowL + 1, 3), wsResumen.cells(startCat - 1, 3)), SortOn:=xlSortOnValues, Order:=xlDescending
        With wsResumen.Sort: .SetRange rngCat: .header = xlYes: .Apply: End With
        
        wsResumen.Range(wsResumen.cells(rowL + 1, 3), wsResumen.cells(startCat - 1, 3)).NumberFormat = "0.00%"
        For r = rowL + 1 To startCat - 1
            If (r - rowL) Mod 2 = 0 Then wsResumen.Range(wsResumen.cells(r, 2), wsResumen.cells(r, 3)).Interior.Color = RGB(248, 250, 252) ' Alternado
            wsResumen.cells(r, 3).HorizontalAlignment = xlCenter
        Next r
        
        rowL = startCat + 2 ' Espacio para la siguiente tarjeta
    End If
    
    ' =======================================================
    ' LIMPIAR [AUTO] PARA LOS RANKINGS
    ' =======================================================
    For r = LastRow To 2 Step -1
        If InStr(1, wsTemp.cells(r, dict("nombre de la campaña")).value, "[AUTO]", vbTextCompare) > 0 Then
            wsTemp.Rows(r).Delete
        End If
    Next r
    LastRow = wsTemp.cells(wsTemp.Rows.Count, "A").End(xlUp).Row
    
    If LastRow >= 2 Then
        Dim limit10 As Integer, limit5 As Integer, limit3 As Integer
        If LastRow - 1 < 10 Then limit10 = LastRow - 1 Else limit10 = 10
        If LastRow - 1 < 5 Then limit5 = LastRow - 1 Else limit5 = 5
        If LastRow - 1 < 3 Then limit3 = LastRow - 1 Else limit3 = 3
        
        ' =======================================================
        ' 4. TARJETA IZQUIERDA 2: TOP 10 APERTURAS
        ' =======================================================
        wsTemp.Sort.SortFields.Clear
        wsTemp.Sort.SortFields.Add key:=wsTemp.Columns(dict("% aperturas")), SortOn:=xlSortOnValues, Order:=xlDescending
        With wsTemp.Sort: .SetRange wsTemp.UsedRange: .header = xlYes: .Apply: End With
        
        wsResumen.cells(rowL, 2).value = "?? TOP " & limit10 & " MAYOR APERTURA"
        wsResumen.cells(rowL, 3).value = "% Apertura"
        
        For r = 1 To limit10
            wsResumen.cells(rowL + r, 2).value = r & ". " & wsTemp.cells(r + 1, dict("nombre de la campaña")).value
            wsResumen.cells(rowL + r, 3).value = wsTemp.cells(r + 1, dict("% aperturas")).value
        Next r
        
        Dim rngTop10 As Range: Set rngTop10 = wsResumen.Range(wsResumen.cells(rowL, 2), wsResumen.cells(rowL + limit10, 3))
        rngTop10.Interior.Color = RGB(255, 255, 255)
        rngTop10.BorderAround xlContinuous, xlThin, , RGB(203, 213, 225)
        With wsResumen.Range(wsResumen.cells(rowL, 2), wsResumen.cells(rowL, 3))
            .Interior.Color = RGB(21, 128, 61) ' Green 700
            .Font.Color = RGB(255, 255, 255): .Font.Bold = True: .Font.Size = 11
        End With
        
        wsResumen.Range(wsResumen.cells(rowL + 1, 3), wsResumen.cells(rowL + limit10, 3)).NumberFormat = "0.00%"
        For r = rowL + 1 To rowL + limit10
            If (r - rowL) Mod 2 = 0 Then wsResumen.Range(wsResumen.cells(r, 2), wsResumen.cells(r, 3)).Interior.Color = RGB(248, 250, 252)
            wsResumen.cells(r, 3).HorizontalAlignment = xlCenter
        Next r
        
        ' =======================================================
        ' 5. TARJETA DERECHA 1: TOP 5 CTOR (INTERACCIÓN)
        ' =======================================================
        wsTemp.Sort.SortFields.Clear
        wsTemp.Sort.SortFields.Add key:=wsTemp.Columns(dict("% tasa de clics por aperturas")), SortOn:=xlSortOnValues, Order:=xlDescending
        With wsTemp.Sort: .SetRange wsTemp.UsedRange: .header = xlYes: .Apply: End With
        
        wsResumen.cells(rowR, 5).value = "?? TOP " & limit5 & " INTERACCIÓN MÁS ALTA"
        wsResumen.cells(rowR, 6).value = "Clics Reales"
        wsResumen.cells(rowR, 7).value = "CTOR %"
        
        For r = 1 To limit5
            wsResumen.cells(rowR + r, 5).value = r & ". " & wsTemp.cells(r + 1, dict("nombre de la campaña")).value
